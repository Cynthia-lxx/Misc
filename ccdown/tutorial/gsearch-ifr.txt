        <input type="text" id="searchQuery" placeholder="输入检索词...">
        <button onclick="search()">检索</button>
    </div>
    <div class="iframe-container" id="iframeContainer" style="display: none;">
        <div class="iframe-title">GitHub</div>
        <iframe id="resultsIframe"></iframe>
    </div>

    <script>
        function search() {
            const query = document.getElementById("searchQuery").value.trim();
            const iframeContainer = document.getElementById("iframeContainer");
            const iframe = document.getElementById("resultsIframe");

            if (query) {
                const githubUrl = `https://github.com/search?q=${encodeURIComponent(query)}`;
                iframe.src = githubUrl;

                // Set iframe height to 30% of the page height
                iframe.onload = function() {
                    try {
                        iframe.style.height = `${Math.min(document.body.clientHeight * 0.3, iframe.contentWindow.document.body.scrollHeight)}px`;
                        iframeContainer.style.display = "block";
                    } catch (error) {
                        // Hide iframe if any loading error occurs
                        iframeContainer.style.display = "none";
                    }
                };

                // Reset height and display the container
                iframe.style.height = "30vh";
                iframeContainer.style.display = "block";
            } else {
                iframeContainer.style.display = "none";
            }
        }
    </script>
